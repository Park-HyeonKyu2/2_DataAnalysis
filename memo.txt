1. requests — 웹페이지 요청

import requests

requests.get()
웹페이지에 GET 요청을 보냄
response = requests.get(url)

headers
브라우저처럼 접속하기 위한 정보
headers = {
    "User-Agent": "Mozilla/5.0"
}
response = requests.get(url, headers=headers)

requests.status_code
응답 상태 코드 확인
print(response.status_code)
- 200: 성공
- 404: 페이지 없음
- 403: 접근 거부
- 500: 서버 오류

response.raise_for_status()
요청에 오류가 있으면 예외 발생

response.text
응답받은 HTML을 문자열로 가져옴
html = response.text

response.json()
JSON 형식의 응답을 파이썬 자료형으로 변환
data = response.json()

8. Selenium 대기
페이지가 로딩될 때까지 기다릴 때 사용.
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

wait = WebDriverWait(driver, 10)
최대 10초 동안 기다림

presence_of_element_located
HTML 안에 요소가 생길 때까지 기다림
element = wait.until(
    EC.presence_of_element_located(
        (By.ID, "top_search")
    )
)

row.find_elements(By.TAG_NAME, "td")
현재 row 안에 있는 모든 <td> 태그를 찾는 코드예요. 😊

여기서 row는 보통 하나의 <tr> 행입니다.

<tr>
    <td>체크박스</td>
    <td>순위</td>
    <td>곡명</td>
    <td>아티스트</td>
    <td>앨범</td>
</tr>
이 행에서 실행하면:

cols = row.find_elements(By.TAG_NAME, "td")
cols에는 <td> 요소들이 리스트처럼 저장됩니다.

cols[0]  # 체크박스 칸
cols[1]  # 순위 칸
cols[2]  # 곡명 칸
cols[3]  # 아티스트 칸
cols[4]  # 앨범 칸
각 요소의 글자를 가져오려면 .text를 사용해요.

title = cols[2].text
artist = cols[3].text
album = cols[4].text
구성별로 보면:

row.find_elements(By.TAG_NAME, "td")
row: 현재 노래 한 줄

find_elements: 조건에 맞는 요소를 전부 찾음

By.TAG_NAME: 태그 이름으로 찾겠다는 뜻

"td": 찾을 태그 이름

참고로 차이는 이렇습니다.

find_element()   # 첫 번째 요소 하나만 찾음
find_elements()  # 조건에 맞는 요소를 모두 찾아 리스트로 반환
즉, 이 코드는 현재 노래 행 안의 모든 열을 가져오는 코드입니다. 🔍


BY_selector하면 셀렉터만 찾는거야?

거의 맞아요. 다만 정확히는 By.CSS_SELECTOR를 쓰면 CSS 선택자 문법으로 요소를 찾는 것이에요. 😊

row.find_elements(By.CSS_SELECTOR, "td")
이 코드는 현재 row 안에서 CSS 선택자 "td"에 해당하는 모든 요소를 찾습니다.

예를 들면:

driver.find_element(By.CSS_SELECTOR, "#top_search")
#top_search → id="top_search"인 요소

driver.find_elements(By.CSS_SELECTOR, ".song")
.song → class="song"인 모든 요소

driver.find_elements(By.CSS_SELECTOR, "tbody tr")
<tbody> 안에 있는 모든 <tr>

row.find_element(By.CSS_SELECTOR, "button.like span.cnt")
button 태그이면서 like 클래스인 요소 안의

span 태그이면서 cnt 클래스인 요소

비교하면:

By.TAG_NAME, "td"
는 태그 이름만 찾고,

By.CSS_SELECTOR, "td:nth-child(3)"
는 CSS 선택자를 사용해서 더 구체적으로 찾을 수 있어요.

참고로 이름은 BY_selector가 아니라 보통 이렇게 씁니다.

By.CSS_SELECTOR
즉, By.CSS_SELECTOR는 셀렉터라는 요소를 찾는 게 아니라, CSS 셀렉터 문법을 이용해 HTML 요소를 찾는 방식입니다. 🔍

from selenium.webdriver.support import expected_conditions as EC
wait = WebDriverWait(driver, 10)
최대 10초 동안 기다림.

presence_of_element_located
HTML 안에 요소가 생길 때까지 기다림.

element = wait.until(
    EC.presence_of_element_located(
        (By.ID, "top_search")
    )
)
화면에 보이지 않아도 HTML에 존재하면 통과.

visibility_of_element_located
요소가 화면에 보일 때까지 기다림.

    EC.visibility_of_element_located(

    )
)

element_to_be_clickable
요소를 클릭할 수 있을 때까지 기다림.

    )
)

presence_of_all_elements_located
여러 요소가 HTML에 생길 때까지 기다림.
rows = wait.until(
    EC.presence_of_all_elements_located(
        (By.CSS_SELECTOR, "tbody tr")
    )

invisibility_of_element_located
요소가 사라질 때까지 기다림.
